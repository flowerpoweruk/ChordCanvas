cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/release/current.json" CC_RELEASE_JSON)
string(JSON CC_VERSION GET "${CC_RELEASE_JSON}" version)
string(JSON CC_BUILD_NUMBER GET "${CC_RELEASE_JSON}" buildNumber)
if(NOT CC_BUILD_NUMBER MATCHES "^[0-9]+$" OR CC_BUILD_NUMBER GREATER 65535)
  message(FATAL_ERROR "Release buildNumber must be a Windows version component (0..65535)")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE CC_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT CC_SOURCE_COMMIT)
  set(CC_SOURCE_COMMIT "unknown")
endif()
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE CC_SOURCE_CHANGES ERROR_QUIET)
if(CC_SOURCE_CHANGES)
  string(APPEND CC_SOURCE_COMMIT "+dirty")
endif()
project(ChordCanvas VERSION ${CC_VERSION} LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MINGW)
  add_link_options(-static)
endif()
add_library(chordcanvas_core Source/Theory/Theory.cpp Source/Commands/Timeline.cpp Source/Export/Midi.cpp Source/Export/TemporaryMidi.cpp Source/Audio/Engine.cpp Source/Persistence/Progression.cpp Source/Model/Session.cpp Source/Diagnostics/Log.cpp)
target_include_directories(chordcanvas_core PUBLIC Source)
target_compile_definitions(chordcanvas_core PUBLIC CC_VERSION="${CC_VERSION}" CC_SOURCE_COMMIT="${CC_SOURCE_COMMIT}" CC_BUILD_NUMBER=${CC_BUILD_NUMBER} CC_BUILD_CONFIGURATION="$<CONFIG>")
target_link_libraries(chordcanvas_core PUBLIC shell32 ole32 version)
add_library(chordcanvas_packaging Source/Packaging/Payload.cpp Source/Packaging/Transaction.cpp Source/Packaging/RegistryImage.cpp Source/Packaging/InstallMetadata.cpp)
target_include_directories(chordcanvas_packaging PUBLIC Source)
target_link_libraries(chordcanvas_packaging PUBLIC bcrypt ole32 advapi32)
target_compile_definitions(chordcanvas_packaging PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
enable_language(RC)
configure_file(Source/Packaging/InstallerVersion.rc.in "${CMAKE_CURRENT_BINARY_DIR}/InstallerVersion.rc" @ONLY)
add_library(chordcanvas_install SHARED Source/Packaging/InstallerApi.cpp "${CMAKE_CURRENT_BINARY_DIR}/InstallerVersion.rc")
set_target_properties(chordcanvas_install PROPERTIES OUTPUT_NAME ChordCanvasInstaller PREFIX "")
target_link_libraries(chordcanvas_install PRIVATE chordcanvas_packaging shell32)
target_compile_definitions(chordcanvas_install PRIVATE CC_VERSION="${CC_VERSION}" NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(chordcanvas_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(chordcanvas_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(core_tests Tests/Unit/CoreTests.cpp)
target_link_libraries(core_tests PRIVATE chordcanvas_core)
add_executable(theory_probe Tests/Integration/TheoryProbe.cpp)
target_link_libraries(theory_probe PRIVATE chordcanvas_core)
add_executable(audio_probe Tests/Integration/AudioProbe.cpp)
target_link_libraries(audio_probe PRIVATE chordcanvas_core)
add_executable(audio_fidelity_tests Tests/Integration/AudioFidelityTests.cpp)
target_link_libraries(audio_fidelity_tests PRIVATE chordcanvas_core)
add_test(NAME audio_fidelity COMMAND audio_fidelity_tests)
set_tests_properties(audio_fidelity PROPERTIES TIMEOUT 30)
add_test(NAME core COMMAND core_tests ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(core PROPERTIES TIMEOUT 180)
add_executable(log_tests Tests/Integration/LogTests.cpp)
target_link_libraries(log_tests PRIVATE chordcanvas_core)
add_test(NAME diagnostics COMMAND log_tests ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(diagnostics PROPERTIES TIMEOUT 30)
add_executable(export_cache_tests Tests/Integration/ExportCacheTests.cpp)
target_link_libraries(export_cache_tests PRIVATE chordcanvas_core)
add_test(NAME export_cache COMMAND export_cache_tests ${CMAKE_CURRENT_BINARY_DIR})
set_tests_properties(export_cache PROPERTIES TIMEOUT 15)
add_executable(package_version_tests Tests/Unit/PackageVersionTests.cpp)
target_include_directories(package_version_tests PRIVATE Source)
add_test(NAME package_version COMMAND package_version_tests)
add_executable(payload_tests Tests/Integration/PayloadTests.cpp)
target_link_libraries(payload_tests PRIVATE chordcanvas_packaging)
add_dependencies(payload_tests core_tests)
add_test(NAME payload COMMAND payload_tests $<TARGET_FILE_DIR:core_tests>)
add_executable(transaction_tests Tests/Integration/TransactionTests.cpp)
target_link_libraries(transaction_tests PRIVATE chordcanvas_packaging)
add_dependencies(transaction_tests core_tests)
add_test(NAME transaction COMMAND transaction_tests $<TARGET_FILE_DIR:core_tests>)
add_executable(registry_tests Tests/Integration/RegistryTests.cpp)
target_link_libraries(registry_tests PRIVATE chordcanvas_packaging)
add_test(NAME registry COMMAND registry_tests ${CMAKE_CURRENT_BINARY_DIR})
add_executable(participant_tests Tests/Integration/ParticipantTests.cpp)
target_link_libraries(participant_tests PRIVATE chordcanvas_packaging)
add_dependencies(participant_tests core_tests)
add_test(NAME participant COMMAND participant_tests $<TARGET_FILE_DIR:core_tests>)
add_executable(install_metadata_tests Tests/Integration/InstallMetadataTests.cpp)
target_link_libraries(install_metadata_tests PRIVATE chordcanvas_packaging)
add_dependencies(install_metadata_tests core_tests)
add_test(NAME install_metadata COMMAND install_metadata_tests $<TARGET_FILE_DIR:core_tests>)
add_executable(installer_api_probe Tests/Integration/InstallerApiProbe.cpp)
target_compile_definitions(installer_api_probe PRIVATE CC_VERSION="${CC_VERSION}")
target_link_libraries(installer_api_probe PRIVATE shell32 ole32)
add_dependencies(installer_api_probe chordcanvas_install)
add_test(NAME installer_api COMMAND installer_api_probe $<TARGET_FILE:chordcanvas_install>)
add_executable(clock_tests Tests/Integration/ClockTests.cpp)
target_link_libraries(clock_tests PRIVATE chordcanvas_core)
add_test(NAME clock COMMAND clock_tests)
add_executable(session_events_tests Tests/Integration/SessionEventsTests.cpp)
target_link_libraries(session_events_tests PRIVATE chordcanvas_core)
add_test(NAME session_events COMMAND session_events_tests)
add_executable(payload_inspect Source/Packaging/PayloadInspect.cpp)
target_link_libraries(payload_inspect PRIVATE chordcanvas_packaging)
if(MINGW)
  target_link_options(payload_inspect PRIVATE -municode)
endif()
option(CHORDCANVAS_BUILD_PLUGIN "Build the ChordCanvas JUCE VST3 instrument" OFF)
if(CHORDCANVAS_BUILD_PLUGIN)
  set(CC_JUCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/_deps/JUCE" CACHE PATH "Pinned JUCE checkout")
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CC_JUCE_PATH}" OUTPUT_VARIABLE CC_JUCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT CC_JUCE_REVISION STREQUAL "72782788ce18c2d4d760b28e0921d6ffc6431102")
    message(FATAL_ERROR "Run Scripts/fetch-dependencies.ps1 to obtain the approved pinned JUCE 9.0.2 checkout")
  endif()
  add_subdirectory("${CC_JUCE_PATH}" "${CMAKE_CURRENT_BINARY_DIR}/JUCE" EXCLUDE_FROM_ALL)
  add_subdirectory(Source/Plugin)
endif()
